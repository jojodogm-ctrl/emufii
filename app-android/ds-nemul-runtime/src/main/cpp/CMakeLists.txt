cmake_minimum_required(VERSION 3.22.1)
project(emufii_nemul_bridge LANGUAGES CXX)
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "The pinned NEmul engine is arm64-v8a only")
endif()
add_library(emufii_nemul_bridge SHARED emufii_nemul_bridge.cpp)
target_compile_features(emufii_nemul_bridge PRIVATE cxx_std_17)
target_compile_options(emufii_nemul_bridge PRIVATE -Wall -Wextra -Werror)
target_link_libraries(emufii_nemul_bridge PRIVATE dl)
